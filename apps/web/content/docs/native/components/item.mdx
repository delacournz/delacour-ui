## Usage

```tsx title="app/index.tsx"
import { Button } from "@/components/ui/button";
import { Icon } from "@/components/ui/icon";
import { Item } from "@/components/ui/item";
import { IconFileText } from "@/lib/icons/central";
import { View } from "react-native";

export default function Example() {
  return (
    <View className="flex-1 justify-center p-4">
      <Item variant="outline">
        <Item.Media variant="icon">
          <Icon icon={IconFileText} />
        </Item.Media>
        <Item.Content>
          <Item.Title>Invoice.pdf</Item.Title>
          <Item.Description>2.4 MB · Updated yesterday</Item.Description>
        </Item.Content>
        <Item.Actions>
          <Button onPress={() => console.log("open")} size="sm" variant="outline">
            <Button.Label>Open</Button.Label>
          </Button>
        </Item.Actions>
      </Item>
    </View>
  );
}
```

## In a ListGroup

An item inside a `ListGroup` takes the group's size, draws no surface of its own and lines up with the group's dividers.

<Preview id="item/in-a-list-group" />

```tsx
import { ListGroup } from "@/components/ui/list-group";

<ListGroup>
  <Item onPress={openWifi}>
    <Item.Media>
      <Icon icon={IconWifiFull} />
    </Item.Media>
    <Item.Content>
      <Item.Title>Wi-Fi</Item.Title>
    </Item.Content>
    <Item.Actions>
      <Icon icon={IconChevronRight} />
    </Item.Actions>
  </Item>
</ListGroup>;
```

## Controls in the actions

<Preview id="item/controls-in-actions" />

```tsx
<Item>
  <Item.Content>
    <Item.Title>Dark mode</Item.Title>
  </Item.Content>
  <Item.Actions>
    <Switch accessibilityLabel="Dark mode" isSelected={isDark} onSelectedChange={setIsDark} />
  </Item.Actions>
</Item>
```

<Callout type="warn">
Do not put a control inside an item that has an `onPress`. A pressable item merges its children into
one accessible element, so VoiceOver cannot reach a button or switch inside it. Leave the item static,
or make the whole row the control with `accessibilityRole="switch"`.
</Callout>

## API Reference

### Item

A `Pressable` when given `onPress` or `onLongPress`, so `feedback`, `haptic` and `busy` are
inherited; a plain view without either.
